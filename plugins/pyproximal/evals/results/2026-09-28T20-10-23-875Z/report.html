import pylops
import pyproximal
from pyproximal.optimization.primal import ProximalGradient
from pylops.utils.metrics import snr

# Problem setup
n = 200
rng = np.random.default_rng(0)

# True model: sparse with 1.0 on specific intervals
xtrue = np.zeros(n)
xtrue[20:60] = 1.0
xtrue[75:80] = 1.0
xtrue[110:130] = 1.0
xtrue[150:185] = 1.0

h /= h.sum()

# ============================================================================
# Formulation: min_x  0.5 * ||Op @ x - y||_2^2 + (lam/2) * ||x||_2^2 + ι_{[0,1]}(x)
#
# where ι_{[0,1]} is the indicator function enforcing 0 ≤ x ≤ 1
#
# Split as: f(x) [smooth] + g(x) [non-smooth]
#   f(x) = 0.5 * ||Op @ x - y||_2^2 + (lam/2) * ||x||_2^2  [data + L2 regularization]
#   g(x) = ι_{[0,1]}(x)  [box constraint, solved via prox/clipping]
#
# Solver: Proximal Gradient with FISTA acceleration
# ============================================================================

# Choose regularization parameter by L-curve or by noise level heuristic
# Here: lam ~ SNR / scale of problem
lam = 0.01  # Strength of smoothness regularization

# Compute operator norm for step size
# L = ||Op||^2 + lam (for the full objective)
print("Computing operator norm for step size...")
L_op = pylops.optimization.eigs.power_iteration(Op.H @ Op, niter=50)[0]
L = L_op + lam  # Upper bound on Lipschitz constant

# Step size: tau <= 1 / L (ProximalGradient needs this for convergence)
tau = 0.95 / L
print(f"  Operator norm squared: {L_op:.6f}")
print(f"  Lipschitz constant L: {L:.6f}")
print(f"  Step size tau: {tau:.6f}")

# Define smooth part: f(x) = 0.5 * ||Op @ x - y||_2^2 + (lam/2) * ||x||_2^2
# We use L2(Op=Op, b=y) for the data term, and handle L2 regularization via custom gradient
class SmoothedObjective:
    """Smooth part of the objective with L2 regularization."""
    def __init__(self, Op, y, lam):
        self.Op = Op
        self.y = y
        self.lam = lam
        self.data_term = pyproximal.L2(Op=Op, b=y)

    def __call__(self, x):
        """Evaluate f(x) = 0.5*||Op@x - y||_2^2 + (lam/2)*||x||_2^2"""
        return self.data_term(x) + (self.lam / 2.0) * np.linalg.norm(x)**2

    def grad(self, x):
        """Gradient: Op.H @ (Op @ x - y) + lam * x"""
        residual = self.Op @ x - self.y
        return self.Op.H @ residual + self.lam * x

g = pyproximal.Box(lower=0.0, upper=1.0)

    """Track objective value at each iteration."""

# Hard rank truncation: keep only top-r singular values
def hard_rank_prox(X, rank):
    """Project to rank-r matrices via SVD."""
    U, s, Vt = np.linalg.svd(X, full_matrices=False)
    s[:rank] = s[:rank]  # keep first rank singular values
    s[rank:] = 0         # zero out the rest
    return U @ np.diag(s) @ Vt

# where r=3 (the true rank)

step_size = 0.1
max_iter = 500
tol = 1e-8

    # Compute gradient of ||Rop(X) - y||_2^2

    grad = (Rop.H @ residual).reshape(n, n)

    X = X - step_size * grad

    # Project to low-rank subspace
    X = hard_rank_prox(X, rank_constraint)

    # Check convergence
    change = np.linalg.norm(X - X_old) / (np.linalg.norm(X_old) + 1e-10)
